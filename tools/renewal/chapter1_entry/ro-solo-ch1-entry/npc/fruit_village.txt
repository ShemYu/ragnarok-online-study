//=   18370, Maysel 18370 -> 18371 with its native rewards, and presentation.
//= Encoding: UTF-8 (Traditional Chinese dialogue; Est/Debris left untranslated).
//= Quest state uses default checkquest as in entry.txt: -1 absent, 1 active
//=   OR inactive, 2 complete; any other value is unknown and refused.
//=   Tuples are (18368,18369,18370,18371); L_State is identical in every
//=   script that reads it (codes as in entry.txt, plus 6 and 7).
//= Native changequest replaces the source quest with the destination; the old
//=   ID is absent afterwards. Command success proves nothing, so every write
//=   is followed by a full readback before anything else happens.
//= Not a transaction: quest, status and inventory are saved separately. A
//=   partial or uncertain result is held for review, never replayed, rolled
//=   back or compensated. No permanent variable of any kind is used.
//= Presentation: per-character cloakonnpc/cloakoffnpc derived from a fresh
//=   tuple (L_Sync). Masks reset on map change and logout; arrival, Shasha
//=   and both final closes resynchronise. Cloaking is not an eligibility
//=   gate: every consequential path checks the quest log itself.
//==============================================================================

// Arrival at the landing cell. Only (1,-1,-1,-1) gets the checked

// every other tuple only resynchronises presentation. Not a warp.
ygg_fruit,73,117,0	script	#ROSOLO_CH1_ARR	HIDDEN_WARP_NPC,1,1,{
	end;

OnTouch:
	.@a = checkquest(18368);
	.@b = checkquest(18369);
	.@c = checkquest(18370);
	.@d = checkquest(18371);

	.@a = checkquest(18368);
	.@b = checkquest(18369);
	.@c = checkquest(18370);
	.@d = checkquest(18371);
	if (.@a == 1 && .@b == -1 && .@c == -1 && .@d == -1) goto L_NotCompleted;
	if (.@a != 2 || .@b != -1 || .@c != -1 || .@d != -1) goto L_Hold;

	.@a = checkquest(18368);
	.@b = checkquest(18369);
	.@c = checkquest(18370);
	.@d = checkquest(18371);

	if (.@a != 2 || .@b != 1 || .@c != -1 || .@d != -1) goto L_Hold;
	dispbottom "任務「天地樹的召喚(2)」已記錄。";
L_Present:
	callsub L_Sync;
	end;

L_NotCompleted:
	debugmes "ROSOLO_CH1: arrival completequest 18368 did not take effect; still (1,-1,-1,-1).";
	dispbottom "紀錄沒有更新，任務維持原狀。";
	end;

L_NotStarted:
	debugmes "ROSOLO_CH1: arrival completed 18368 but setquest 18369 did not take effect; holding (2,-1,-1,-1).";
	dispbottom "紀錄只更新了一半，不會自動修復。";
	end;

L_Hold:
	debugmes "ROSOLO_CH1: unexpected arrival readback 18368=" + .@a + " 18369=" + .@b + " 18370=" + .@c + " 18371=" + .@d + "; stopped, no further change.";
	dispbottom "任務紀錄出現無法確認的變化。";
	end;

L_Sync:
	if (strcharinfo(3) != "ygg_fruit") return;
	.@s = callsub(L_State);
	.@cid = getcharid(0);
	if (.@s == 4) {
		cloakonnpc "梅雪#ROSOLO_CH1_MAY",.@cid;
		cloakoffnpc "Est#ROSOLO_CH1_EST",.@cid;
		cloakoffnpc "Debris#ROSOLO_CH1_DEB",.@cid;
		return;
	}
	if (.@s == 6) {
		cloakoffnpc "梅雪#ROSOLO_CH1_MAY",.@cid;
		cloakonnpc "Est#ROSOLO_CH1_EST",.@cid;
		cloakoffnpc "Debris#ROSOLO_CH1_DEB",.@cid;
		return;
	}
	if (.@s == 7) {
		cloakoffnpc "梅雪#ROSOLO_CH1_MAY",.@cid;
		cloakonnpc "Est#ROSOLO_CH1_EST",.@cid;
		cloakonnpc "Debris#ROSOLO_CH1_DEB",.@cid;
		return;
	}
	return;

L_State:
	.@a = checkquest(18368);
	.@b = checkquest(18369);
	.@c = checkquest(18370);
	.@d = checkquest(18371);
	if (.@a != -1 && .@a != 1 && .@a != 2) return 3;
	if (.@b != -1 && .@b != 1 && .@b != 2) return 3;
	if (.@c != -1 && .@c != 1 && .@c != 2) return 3;
	if (.@d != -1 && .@d != 1 && .@d != 2) return 3;
	if (.@a == -1 && .@b == -1 && .@c == -1 && .@d == -1) return 0;
	if (.@a == 1) {
		if (.@b == -1 && .@c == -1 && .@d == -1) return 1;
		return 3;
	}
	if (.@b == 1 && (.@c != -1 || .@d != -1)) return 3;
	if (.@c == 1 && .@d != -1) return 3;
	if (.@a == 2 && .@b == 1 && .@c == -1 && .@d == -1) return 4;
	if (.@a == 2 && .@b == -1 && .@c == -1 && .@d == -1) return 5;
	if (.@a == 2 && .@b == -1 && .@c == 1 && .@d == -1) return 6;
	if (.@a == 2 && .@b == -1 && .@c == -1 && .@d == 1) return 7;
	return 2;
}
